        let styleEl = null;
        let observer = null;

        function esconderSugestaoResposta() {
            document.querySelectorAll(SUGESTAO_SELECTOR).forEach(btn => {
                btn.style.display = 'none';
            });
        }

        // Adicionado: sem isso o destroy() não desfazia o que o init() escondia
        function mostrarSugestaoResposta() {
            document.querySelectorAll(SUGESTAO_SELECTOR).forEach(btn => {
                btn.style.display = '';
            });
        }

        function inserirTexto(texto, btn) {
            const editor = document.querySelector(EDITOR_SELECTOR);
            if (!editor) return;

            editor.innerHTML = texto
                .replace(/\r/g, '')
                .split('\n')
                .map(l => l.trim() === '' ? '<p><br></p>' : `<p>${l}</p>`)
                .join('');

            editor.dispatchEvent(new Event('input', { bubbles: true }));

            const original = btn.innerHTML;
            btn.innerHTML = '✅';
            setTimeout(() => btn.innerHTML = original, 800);
        }

        function inserirBotoes() {
            const toolbar = document.querySelector(TOOLBAR_SELECTOR);
            if (!toolbar || document.getElementById(GRUPO_ID)) return;

            const grupo = document.createElement('div');
            grupo.id = GRUPO_ID;
            grupo.style.cssText = `display:flex;gap:3px;margin-right:15px;`;

            MENSAGENS.forEach(msg => {
                const btn = document.createElement('button');
                btn.type = 'button';
                btn.innerHTML = msg.icon;
                btn.setAttribute('data-tooltip', msg.title);
                btn.style.cssText = `
                    width:28px;height:28px;background:transparent;border:none;
                    border-radius:6px;font-size:16px;cursor:pointer;opacity:.75;
                    display:flex;align-items:center;justify-content:center;
                `;
                btn.onmouseenter = () => btn.style.opacity = '1';
                btn.onmouseleave = () => btn.style.opacity = '.75';
                btn.onclick = () => inserirTexto(msg.texto, btn);
                grupo.appendChild(btn);
            });

            toolbar.insertBefore(grupo, toolbar.firstChild);
        }

        return {
            id: 'tallos_botoes_mensagens',
            name: 'TALLOS - Barras e mensagens',
            defaultOn: true,
            fixo: true, // módulo embutido no código-fonte, não pode ser excluído pelo painel

            init() {
                // este widget só faz sentido no chat do Tallos (era @match .../app/chat*)
                if (!location.hostname.includes('tallos.com.br') || !location.pathname.includes('/app/chat')) return;

                styleEl = document.createElement('style');
                styleEl.innerHTML = `
                    #${GRUPO_ID} button { position: relative; }
                    #${GRUPO_ID} button::after {
                        content: attr(data-tooltip);
                        position: absolute;
                        top: -34px;
                        left: 50%;
                        transform: translateX(-50%);
                        background: #000;
                        color: #fff;
                        padding: 4px 8px;
                        border-radius: 6px;
                        font-size: 11px;
                        white-space: nowrap;
                        opacity: 0;
                        pointer-events: none;
                        transition: opacity .15s ease;
                        z-index: 9999;
                    }
                    #${GRUPO_ID} button:hover::after { opacity: 1; }
                `;
                document.head.appendChild(styleEl);

                observer = new MutationObserver(() => {
                    esconderSugestaoResposta();
                    inserirBotoes();
                });
                observer.observe(document.body, { childList: true, subtree: true });

                esconderSugestaoResposta();
                inserirBotoes();
            },

            destroy() {
                if (observer) {
                    observer.disconnect();
                    observer = null;
                }
                document.getElementById(GRUPO_ID)?.remove();
                styleEl?.remove();
                styleEl = null;
                mostrarSugestaoResposta();
            }
        };
    }

    registrarModulo(criarModuloTallosBotoesMensagens());
