// ==UserScript==
// @name         Botão Preencher Email
// @version      1.5.0
// @description  Botão Preencher Email
// @match        *://*.smarttravelit.com/*
// @grupo        SmartBus
// ==/UserScript==

// Este script é um módulo: registrarModulo() liga e desliga o módulo pelo hub,
// sem precisar recarregar a página.
// =========================================================================
// MÓDULO 11 — Botão Preencher Email
// =========================================================================
function criarModuloSmartBusPreencherEmail() {
    const EMAIL = 'gilmario12.2025@gmail.com';

    const BOTAO_ID = 'btn-preencher-email';

    let observer = null;

    function setNativeValue(input, value) {
        const setter = Object.getOwnPropertyDescriptor(
            HTMLInputElement.prototype,
            'value'
        ).set;

        setter.call(input, value);

        input.dispatchEvent(new Event('input', { bubbles: true }));
        input.dispatchEvent(new Event('change', { bubbles: true }));
    }

    function injectEmailButton() {
        const div = document.querySelector('.row.row-form-btns');

        if (!div) return;
        if (document.getElementById(BOTAO_ID)) return;

        div.style.display = 'flex';
        div.style.alignItems = 'center';

        const btn = document.createElement('button');
        btn.id = BOTAO_ID;
        btn.type = 'button';
        btn.textContent = 'Preencher email';

        Object.assign(btn.style, {
            background: '#2563eb',
            color: '#fff',
            border: 'none',
            borderRadius: '4px',
            padding: '10px 18px',
            cursor: 'pointer',
            fontWeight: '600',
            marginLeft: '10px'
        });

        btn.addEventListener('click', () => {
            const lightbox = document.querySelector('.div-lightbox-content');

            if (!lightbox) {
                alert('Lightbox não encontrada!');
                return;
            }

            const campoEmail = lightbox.querySelector('input.input');

            if (!campoEmail) {
                alert('Campo de e-mail não encontrado!');
                return;
            }

            setNativeValue(campoEmail, EMAIL);
            campoEmail.focus();
        });

        const btnEnviar = [...div.querySelectorAll('button')]
            .find(b => b.textContent.trim().toUpperCase().includes('ENVIAR'));

        if (btnEnviar) {
            div.insertBefore(btn, btnEnviar);
        } else {
            div.appendChild(btn);
        }
    }

    return {
        id: 'preencher-email',
        name: 'SMART BUS - Botão Preencher Email',
        defaultOn: true,
        fixo: true,

        init() {
            observer = new MutationObserver(injectEmailButton);

            observer.observe(document.body, {
                childList: true,
                subtree: true
            });

            injectEmailButton();
        },

        destroy() {
            observer?.disconnect();
            observer = null;

            document.getElementById(BOTAO_ID)?.remove();
        }
    };
}

registrarModulo(criarModuloSmartBusPreencherEmail());
