// ==UserScript==
// @name         Horário + Serviço
// @version      1.5.0
// @description  Horário + Serviço (+ Print de Poltronas)
// @match        *://*.smarttravelit.com/*
// @grupo        SmartBus
// ==/UserScript==

// Este script é um módulo: registrarModulo() liga e desliga o módulo pelo hub,
// sem precisar recarregar a página.
    // =========================================================================
// MÓDULO 12 — Horário + Serviço (+ Print de Poltronas)
// Requer no cabeçalho do Painel (bloco ==UserScript==):
// // @require https://cdnjs.cloudflare.com/ajax/libs/html2canvas/1.4.1/html2canvas.min.js
// =========================================================================
function criarModuloHorarioServico() {

    let observer = null;
    let interval = null;
    let style = null;

    const TEXTO_BOTAO = "Print poltronas";
    const COR_BOTAO = "darkred";

    const normalizar = t => (t || "").replace(/\s+/g, "").toUpperCase();

    function criarSpan(texto, cor) {
        const s = document.createElement("span");
        s.textContent = texto;
        if (cor) s.style.color = cor;
        return s;
    }

    // Sobe a partir do título até o card do mapa, parando ANTES do bloco
    // que também contém o card de TARIFAS (não depende de ter poltronas livres)
    function acharMapa(titulo) {
        const ehLimite = el =>
            el === document.body ||
            el.tagName === "APP-SERVICEROW" ||
            /TARIFAS|SEGURO OPCIONAL|ESCOLHA UMA TARIFA/i.test(el.textContent || "");

        let el = titulo.closest(".seat-map-route") || titulo;
        while (el.parentElement && !ehLimite(el.parentElement)) {
            el = el.parentElement;
        }
        return el;
    }

    function feedback(btn, msg, erro = false) {
        btn.textContent = msg;
        btn.style.background = erro ? "#dc2626" : "#16a34a";
        setTimeout(() => {
            btn.textContent = TEXTO_BOTAO;
            btn.style.background = COR_BOTAO;
            btn.disabled = false;
        }, 1800);
    }

    function nomeArquivo(titulo, horario) {
        const rota = ((titulo.innerText || "").split("\n")[0] || "")
            .replace(/TRECHO\s*:/i, "")
            .normalize("NFD").replace(/[\u0300-\u036f]/g, "")
            .replace(/[^A-Za-z0-9]+/g, "-")
            .replace(/^-|-$/g, "");
        return `poltronas_${rota}_${horario.replace(":", "h")}.png`;
    }

    // Cor de fundo efetiva (sobe até 3 níveis se for transparente)
    function fundo(win, el) {
        for (let i = 0; el && i < 4; i++, el = el.parentElement) {
            const m = win.getComputedStyle(el).backgroundColor.match(/\d+/g);
            if (m && !(m.length === 4 && +m[3] === 0)) return m.map(Number);
        }
        return null;
    }

    function prepararClone(doc) {
        const win = doc.defaultView;

        // percorre só o mapa clonado, não a página inteira
        const raiz = doc.querySelector("[data-tm-mapa]") || doc.body;

        raiz.querySelectorAll("*").forEach(el => {

            // Botão de print some
            if (el.classList.contains("tm-print-btn")) {
                el.style.visibility = "hidden";
                return;
            }

            // Compara só o texto PRÓPRIO do elemento
            const proprio = normalizar(
                [...el.childNodes].filter(n => n.nodeType === 3).map(n => n.nodeValue).join("") ||
                el.value || ""
            );

            // "SEM ASSENTO" + botão "+": esconde o grupo inteiro
            if (proprio === "SEMASSENTO") {
                let alvo = el.closest("button, a, [role=button]") || el;
                while (
                    alvo.parentElement &&
                    normalizar(alvo.parentElement.textContent).startsWith("SEMASSENTO") &&
                    normalizar(alvo.parentElement.textContent).length < 25
                ) {
                    alvo = alvo.parentElement;
                }
                alvo.style.visibility = "hidden";
                return;
            }

            // "Exibir legenda" / "Ocultar legenda"
            if (proprio === "EXIBIRLEGENDA" || proprio === "OCULTARLEGENDA") {
                (el.closest("button, a, [role=button]") || el).style.visibility = "hidden";
                return;
            }

            // Poltronas vermelhas (ocupadas) sem número
            if (!el.children.length && /^\d{1,2}$/.test(el.textContent.trim())) {
                const c = fundo(win, el);
                if (c && c[0] > 180 && c[1] < 130 && c[2] < 130) {
                    el.style.color = "transparent";
                    el.style.textShadow = "none";
                }
            }
        });
    }

    // Carrega o html2canvas sob demanda se o @require não estiver disponível
    const URL_H2C = "https://cdnjs.cloudflare.com/ajax/libs/html2canvas/1.4.1/html2canvas.min.js";
    let promessaH2c = null;

    function pegarHtml2canvas() {
        if (typeof html2canvas !== "undefined") return html2canvas;
        if (window.html2canvas) return window.html2canvas;
        if (typeof unsafeWindow !== "undefined" && unsafeWindow.html2canvas) return unsafeWindow.html2canvas;
        return null;
    }

    function carregarHtml2canvas() {
        const pronto = pegarHtml2canvas();
        if (pronto) return Promise.resolve(pronto);

        if (!promessaH2c) {
            promessaH2c = new Promise((resolve, reject) => {
                const pagina = typeof unsafeWindow !== "undefined" ? unsafeWindow : window;

                // Se o site usar AMD (define.amd), o html2canvas não cria a global:
                // desativa o define só durante o carregamento
                const defineOriginal = pagina.define;
                if (defineOriginal?.amd) pagina.define = undefined;
                const restaurar = () => { if (defineOriginal?.amd) pagina.define = defineOriginal; };

                const s = document.createElement("script");
                s.src = URL_H2C;
                s.onload = () => {
                    restaurar();
                    const f = pegarHtml2canvas();
                    f ? resolve(f) : reject(new Error("h2c"));
                };
                s.onerror = () => { restaurar(); reject(new Error("h2c")); };
                document.head.appendChild(s);
            }).catch(e => {
                promessaH2c = null; // permite tentar de novo no próximo clique
                throw e;
            });
        }
        return promessaH2c;
    }

    // Elementos que não entram na conta da largura (botão de print, "SEM ASSENTO", legenda)
    function ignorarNaLargura(el) {
        if (el.closest(".tm-print-btn")) return true;
        const grupo = el.closest("button, a, [role=button]") || el;
        return /SEM\s*ASSENTO|LEGENDA/i.test(grupo.textContent || "");
    }

    // Poltrona = quadradinho visível (com ou sem número)
    function ehAssento(el) {
        const r = el.getBoundingClientRect();
        return r.width >= 20 && r.width <= 90 && Math.abs(r.width - r.height) <= 15;
    }

    // Recorte do print: largura até a traseira do ônibus e altura até a parte
    // de baixo do ônibus, sem nunca cortar o texto do TRECHO / HORÁRIO
    function medirRecorte(mapa, titulo) {
        const rMapa = mapa.getBoundingClientRect();
        const inteiro = { largura: Math.ceil(rMapa.width), altura: Math.ceil(rMapa.height) };
        const limite = rMapa.width * 0.95; // ignora wrappers que ocupam o card inteiro
        let direita = 0;
        let base = 0;

        // 1) Fim do texto do título (TRECHO e HORÁRIO), medido pelo texto real
        const walker = document.createTreeWalker(titulo, NodeFilter.SHOW_TEXT);
        const range = document.createRange();
        let n;
        while ((n = walker.nextNode())) {
            if (!n.nodeValue.trim() || ignorarNaLargura(n.parentElement)) continue;
            range.selectNodeContents(n);
            const r = range.getBoundingClientRect();
            if (r.width > 0) {
                direita = Math.max(direita, r.right);
                base = Math.max(base, r.bottom);
            }
        }

        // 2) Faixa vertical do ônibus, a partir das poltronas
        const pecas = [...mapa.querySelectorAll("*")].filter(
            el => !titulo.contains(el) && !ignorarNaLargura(el)
        );
        const assentos = pecas.filter(ehAssento);
        if (!assentos.length) return inteiro;

        const rects = assentos.map(a => a.getBoundingClientRect());
        const faixaTopo = Math.min(...rects.map(r => r.top));
        const faixaBase = Math.max(...rects.map(r => r.bottom));

        // 3) Peças do desenho na altura do ônibus (lataria, traseira, poltronas).
        //    O que está só acima ou só abaixo ("Exibir legenda", "Somente mais X
        //    poltronas", "BJS → JDO") fica de fora.
        pecas.forEach(el => {
            const r = el.getBoundingClientRect();
            if (r.width <= 0 || r.height <= 0 || r.width >= limite) return;
            if (r.bottom <= faixaTopo || r.top >= faixaBase) return;
            direita = Math.max(direita, r.right);
            base = Math.max(base, r.bottom);
        });

        const MARGEM_LADO = 10;  // respiro depois da traseira
        const MARGEM_BAIXO = 20; // respiro abaixo do ônibus
        return {
            largura: Math.min(inteiro.largura, Math.ceil(direita - rMapa.left + MARGEM_LADO)),
            altura: Math.min(inteiro.altura, Math.ceil(base - rMapa.top + MARGEM_BAIXO))
        };
    }

    async function tirarPrint(btn, titulo, horario) {

        const mapa = acharMapa(titulo);
        if (!mapa) return feedback(btn, "Mapa não encontrado", true);

        btn.disabled = true;
        btn.textContent = "Gerando...";

        const { largura, altura } = medirRecorte(mapa, titulo);
        mapa.setAttribute("data-tm-mapa", "1");

        const gerarBlob = carregarHtml2canvas().then(h2c => h2c(mapa, {
            backgroundColor: "#ffffff",
            width: largura,  // corta logo depois da traseira do ônibus
            height: altura,  // corta logo abaixo do ônibus
            imageTimeout: 1500,
            // não clona/carrega o que está fora do mapa (outras viagens, imagens da página)
            ignoreElements: el =>
                !mapa.contains(el) && !el.contains(mapa) &&
                (el.tagName === "APP-SERVICEROW" ||
                 /^(IMG|IFRAME|VIDEO|CANVAS|OBJECT|EMBED)$/.test(el.tagName)),
            scale: 2,
            useCORS: true,
            logging: false,
            // Mexe só na cópia usada pelo print, a tela real não muda
            onclone: doc => prepararClone(doc)
        })).then(c => new Promise(res => c.toBlob(res, "image/png")))
           .finally(() => mapa.removeAttribute("data-tm-mapa"));

        try {
            // Promise direto no ClipboardItem: funciona no Chrome e no Safari
            await navigator.clipboard.write([new ClipboardItem({ "image/png": gerarBlob })]);
            feedback(btn, "✓ Copiado!");
        } catch (e) {
            let blob = null;
            try {
                blob = await gerarBlob;
            } catch (erro) {
                console.error("[Print poltronas]", erro);
                return feedback(btn, erro?.message === "h2c" ? "html2canvas não carregou" : "Erro no print", true);
            }
            if (!blob) return feedback(btn, "Erro no print", true);
            const a = document.createElement("a");
            a.href = URL.createObjectURL(blob);
            a.download = nomeArquivo(titulo, horario);
            a.click();
            setTimeout(() => URL.revokeObjectURL(a.href), 2000);
            feedback(btn, "✓ Baixado");
        }
    }

    function criarBotao(titulo, horario) {
        const b = document.createElement("button");
        b.type = "button";
        b.className = "tm-print-btn";
        b.textContent = TEXTO_BOTAO;
        b.setAttribute("data-html2canvas-ignore", "true");
        Object.assign(b.style, {
            marginLeft: "300px",
            padding: "8px 12px",
            border: "none",
            borderRadius: "4px",
            background: COR_BOTAO,
            color: "#fff",
            fontSize: "11px",
            fontWeight: "700",
            cursor: "pointer"
        });
        b.addEventListener("click", e => {
            e.preventDefault();
            e.stopPropagation();
            tirarPrint(b, titulo, horario);
        });
        return b;
    }

    function atualizar() {

        document.querySelectorAll("app-servicerow").forEach(servicerow => {

            const titulo = servicerow.querySelector(".seat-map-route h4");
            const horarioCol = servicerow.querySelector(".service-info .service-col-3");
            const servico = servicerow.querySelector(".service-info .service-col-7 span");

            if (!titulo || !horarioCol || !servico) return;

            const textoOriginal = horarioCol.textContent
                .replace(/\s+/g, " ")
                .trim();

            const primeiroHorario = (textoOriginal.match(/\d{2}:\d{2}/) || [""])[0];
            const nomeServico = servico.textContent.trim();
            const chave = `${primeiroHorario}|${nomeServico}`;

            // wrap permite a info ir para a linha de baixo
            titulo.style.display = "flex";
            titulo.style.alignItems = "center";
            titulo.style.flexWrap = "wrap";

            // span (não div) por causa do CSS ".seat-map-route h4 > div { display:none }"
            let info = titulo.querySelector(".tm-info");

            if (!info) {

                info = document.createElement("span");
                info.className = "tm-info";

                Object.assign(info.style, {
                    flexBasis: "100%",
                    display: "flex",
                    alignItems: "center",
                    gap: "6px",
                    marginTop: "4px",
                    whiteSpace: "nowrap",
                    fontWeight: "700",
                    fontSize: "inherit",
                    color: "inherit"
                });

                titulo.appendChild(info);
            }

            if (info.dataset.chave === chave) return;
            info.dataset.chave = chave;

            // cor do label "TRECHO:" para os labels novos
            const lbl = [...titulo.children].find(
                c => c !== info && /TRECHO/i.test(c.textContent)
            ) || titulo;
            const corLabel = getComputedStyle(lbl).color;

            info.textContent = "";
            info.append(
                criarSpan("HORÁRIO:", corLabel),
                criarSpan(primeiroHorario),
                criarSpan("|", corLabel),
                criarSpan(nomeServico),
                criarBotao(titulo, primeiroHorario)
            );

        });

    }

    return {

        id: "horario-servico",

        name: "SMART BUS - Horário + Serviço",

        defaultOn: true,

        fixo: false,

        init() {

            style = document.createElement("style");

            style.textContent = `
                .seat-map-route h4 > div{
                    display:none !important;
                }
            `;

            document.head.appendChild(style);

            atualizar();

            observer = new MutationObserver(atualizar);

            observer.observe(document.body, {
                childList: true,
                subtree: true
            });

            interval = setInterval(atualizar, 1000);

            // deixa o html2canvas pronto antes do primeiro clique
            setTimeout(() => carregarHtml2canvas().catch(() => {}), 1500);

        },

        destroy() {

            observer?.disconnect();

            observer = null;

            clearInterval(interval);

            interval = null;

            style?.remove();

            style = null;

            document.querySelectorAll(".tm-info").forEach(e => e.remove());

        }

    };

}

registrarModulo(criarModuloHorarioServico());
